// V5 = V4.1 + Météo pondérée (40% pluie, 15km/h vent)
// Données réelles Chantilly 40% pluie = bon souple -> avantage 15
export function computeV5(apifyData, oddsData, meteoData){
  const horses = [...Array(16).keys()].map(n=>n+1).map(num=>{
    let score = 50;
    // Apify: forme, ferrure, corde
    const f = apifyData.find(x=>x.num==num);
    if(f){ score += f.forme*2; score += f.deferre?3:0; score += f.corde<=5?2:0; }
    // Oddspapi: drift 7h->10h (cote qui baisse = chaud)
    const o = oddsData.find(x=>x.num==num);
    if(o && o.drift<0) score += 5;
    // Météo V5: pluie + vent
    if(meteoData.pluie24h>10 && [15,14,5].includes(num)) score += 8; // nageurs
    if(meteoData.vent>12 && f && f.corde>10) score -= 3; // vent exterieur
    if(num==15) score += 10; // base Chantilly 1400 PSF
    return {num, score};
  }).sort((a,b)=>b.score-a.score);

  const principal = horses.slice(0,5).map(h=>h.num); // ex: [15,4,14,5,7]
  const surprise = [horses[0].num, horses[5].num, horses[6].num, horses[2].num, horses[7].num]; // ex: [15,6,12,8,18] - on pioche outsiders météo
  const outsiders = horses.slice(5,8).map(h=>h.num);
  return {principal, surprise, outsiders, scores:horses, meteo:meteoData};
}
