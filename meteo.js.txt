import fetch from 'node-fetch';
export async function getMeteoChantilly(){
  // OpenWeather One Call - Chantilly 49.1946, 2.4712
  const key = process.env.OPENWEATHER_KEY;
  const url = `https://api.openweathermap.org/data/2.5/weather?lat=49.1946&lon=2.4712&appid=${key}&units=metric&lang=fr`;
  const r = await fetch(url).then(x=>x.json());
  const pluie24h = r.rain?.['1h'] ? r.rain['1h']*24 : 8; // approx
  const vent = r.wind?.speed ? r.wind.speed*3.6 : 15; // km/h
  const etat = pluie24h>15 ? 'Bon souple' : 'Bon';
  return {pluie24h, vent, etat, temp:r.main.temp, desc:r.weather[0].description};
}
